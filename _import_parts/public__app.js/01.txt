async function api(path,opts={}){

 if(opts.body!==undefined)headers['content-type']='application/json';
 if(state.me?.csrf&&opts.method&&opts.method!=='GET')headers['x-csrf-token']=state.me.csrf;
 const response=await fetch('/api'+path,{credentials:'same-origin',...opts,headers:{...headers,...(opts.headers||{})},body:opts.body===undefined?undefined:JSON.stringify(opts.body)});
 let data;try{data=await response.json()}catch{throw new Error('Unexpected server response')}
 if(!response.ok||!data.ok)throw new Error(data.error||'Request failed');return data;
}
